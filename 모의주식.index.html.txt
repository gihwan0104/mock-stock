    // 한국거래소 호가 단위(Tick) 계산
    function getTickSize(price) {
        if (price < 2000) return 1;
        if (price < 5000) return 5;
        if (price < 20000) return 10;
        if (price < 50000) return 50;
        if (price < 200000) return 100;
        if (price < 500000) return 500;
        return 1000;
    }

    function alignToTick(price) {
        const tick = getTickSize(price);
        return Math.max(tick, Math.round(price / tick) * tick);
    }

    // 상한가(+50%) 및 하한가(-50%) 계산
    function getPriceLimits(basePrice) {
        return {
            upper: alignToTick(basePrice * 1.5),
            lower: alignToTick(basePrice * 0.5)
        };
    }

    // 초기 대칭형 캔들 생성 (50:50 확률 + 상하한가 범위 준수)
    function createInitialCandles(basePrice, count) {
        const candles = [];
        const limits = getPriceLimits(basePrice);
        let price = basePrice;

        for (let i = 0; i < count; i++) {
            const open = price;
            let high = open;
            let low = open;

            for (let step = 0; step < 6; step++) {
                const tick = getTickSize(price);
                const dir = Math.random() < 0.5 ? 1 : -1; // 정확히 50:50 대칭 확률
                let nextP = price + dir * tick;
                if (nextP > limits.upper) nextP = limits.upper;
                if (nextP < limits.lower) nextP = limits.lower;
                price = nextP;
                if (price > high) high = price;
                if (price < low) low = price;
            }
            candles.push({ open, high, low, close: price });
        }
        return candles;
    }

    // 호가창 10단계 생성 (상·하한가 초과 호가는 잔량 0 처리)
    function buildOrderBook(price, basePrice) {
        const tick = getTickSize(price);
        const limits = getPriceLimits(basePrice);
        const asks = [];
        const bids = [];

        for (let i = 5; i >= 1; i--) {
            const p = price + i * tick;
            asks.push({
                price: p,
                qty: p > limits.upper ? 0 : Math.floor(Math.random() * 450) + 40
            });
        }
        for (let i = 1; i <= 5; i++) {
            const p = Math.max(tick, price - i * tick);
            bids.push({
                price: p,
                qty: p < limits.lower ? 0 : Math.floor(Math.random() * 450) + 40
            });
        }
        return { asks, bids };
    }

    async function initCloudDatabase() {

            const marketDocRef = doc(db, 'artifacts', appId, 'public', 'data', 'market', 'state');

        }

    function isMarketOpenNow() {
        if (nightMarketOverride) return true;
        const now = new Date();
        const hour = now.getHours();
        // 06:00 이상 ~ 23:59 이하 거래 가능 (00:00 ~ 05:59 장 마감)
        return hour >= 6 && hour <= 23;
    }

        }
    }

    // 1초 주기 대칭 시뮬레이션
    function stepMarketSimulation() {
        updateClockAndMarketStatus();
        if (!isMarketOpenNow()) return;

        tickCounter++;
        Object.values(stocks).forEach(st => {
            const tick = getTickSize(st.price);
            const limits = getPriceLimits(st.prevClose);

            // 정확히 대칭 확률: 35% +1틱 상승, 35% -1틱 하락, 30% 보합
            const r = Math.random();
            let delta = 0;
            if (r < 0.35) delta = 1;
            else if (r < 0.70) delta = -1;

            let nextPrice = st.price + delta * tick;
            // ±50% 상·하한가 엄격 제한
            if (nextPrice > limits.upper) nextPrice = limits.upper;
            if (nextPrice < limits.lower) nextPrice = limits.lower;

            st.price = nextPrice;

            // 모든 타임프레임의 현재 봉 갱신
            TIMEFRAMES.forEach(tf => {
                const list = st.candles[tf];
                const last = list[list.length - 1];
                last.close = nextPrice;
                if (nextPrice > last.high) last.high = nextPrice;
                if (nextPrice < last.low) last.low = nextPrice;
            });

            // 10초마다 5분봉 새 캔들 추가
            if (tickCounter % 10 === 0) {
                const c5 = st.candles['5m'];
                c5.push({ open: nextPrice, high: nextPrice, low: nextPrice, close: nextPrice });
                if (c5.length > 50) c5.shift();
            }

            st.orderBook = buildOrderBook(nextPrice, st.prevClose);
        });

        refreshAllViews();

        // 4초마다 클라우드에 시장 상태 백업
        if (tickCounter % 4 === 0) {
            saveMarketStateToCloud();
        }
    }

    /***********************************************************************
     * 5. 화면 렌더링 (로비 리더보드, 10개 기업 카드, 3단 거래화면)
     ***********************************************************************/
    function getAccountTotalValue(acc) {
        let stockEval = 0;
        Object.entries(acc.holdings || {}).forEach(([sid, h]) => {
            if (h && h.qty > 0 && stocks[sid]) {
                stockEval += h.qty * stocks[sid].price;
            }
        });
        return {
            cash: acc.cash,
            stockEval,
            total: acc.cash + stockEval
        };
    }

    function renderLeaderboard() {

            return { nickname: acc.nickname, cash, stockEval, total, rate };

        if (list.length === 0) {

    function drawCandleChart() {
        const canvas = document.getElementById('stockCandleCanvas');
        const container = document.getElementById('chartContainer');
        if (!canvas || !container || currentScreen !== 'TRADE') return;

        const dpr = window.devicePixelRatio || 1;
        const rect = container.getBoundingClientRect();
        canvas.width = rect.width * dpr;
        canvas.height = rect.height * dpr;

        const ctx = canvas.getContext('2d');
        ctx.scale(dpr, dpr);

        const W = rect.width;
        const H = rect.height;
        ctx.fillStyle = '#0b0e14';
        ctx.fillRect(0, 0, W, H);

        const st = stocks[selectedStockId];
        const candles = st.candles[selectedTimeframe] || [];
        if (candles.length === 0) return;

        const padTop = 24, padBottom = 24, padLeft = 16, padRight = 72;
        const plotW = W - padLeft - padRight;
        const plotH = H - padTop - padBottom;

        let minP = Infinity, maxP = -Infinity;
        candles.forEach(c => {
            if (c.low < minP) minP = c.low;
            if (c.high > maxP) maxP = c.high;
        });
        const margin = Math.max((maxP - minP) * 0.12, getTickSize(st.price) * 4);
        minP = Math.max(0, minP - margin);
        maxP += margin;

        // 배경 수평 그리드 및 우측 가격 레이블
        ctx.strokeStyle = '#1e293b';
        ctx.fillStyle = '#64748b';
        ctx.font = '11px Pretendard, sans-serif';
        ctx.textAlign = 'left';

        for (let i = 0; i <= 4; i++) {
            const y = padTop + (plotH / 4) * i;
            const pVal = Math.round(maxP - ((maxP - minP) / 4) * i);
            ctx.beginPath();
            ctx.moveTo(padLeft, y);
            ctx.lineTo(padLeft + plotW, y);
            ctx.stroke();
            ctx.fillText(pVal.toLocaleString(), padLeft + plotW + 8, y + 4);
        }

        // 캔들 그리기
        const stepW = plotW / candles.length;
        const bodyW = Math.max(3, Math.min(16, stepW * 0.65));

        candles.forEach((c, idx) => {
            const cx = padLeft + idx * stepW + stepW / 2;
            const yHigh = padTop + ((maxP - c.high) / (maxP - minP)) * plotH;
            const yLow = padTop + ((maxP - c.low) / (maxP - minP)) * plotH;
            const yOpen = padTop + ((maxP - c.open) / (maxP - minP)) * plotH;
            const yClose = padTop + ((maxP - c.close) / (maxP - minP)) * plotH;

            const isUp = c.close >= c.open;
            const color = isUp ? '#f04452' : '#3182f6';

            ctx.strokeStyle = color;
            ctx.lineWidth = 1.2;
            ctx.beginPath();
            ctx.moveTo(cx, yHigh);
            ctx.lineTo(cx, yLow);
            ctx.stroke();

            ctx.fillStyle = color;
            const topY = Math.min(yOpen, yClose);
            const height = Math.max(2, Math.abs(yClose - yOpen));
            ctx.fillRect(cx - bodyW / 2, topY, bodyW, height);
        });
    }

    window.submitTradeOrder = async function() {
        if (!currentAccountKey || !allAccounts[currentAccountKey]) {
            showToast('거래하려면 먼저 로그인(또는 계정 생성)을 해주세요!', 'error');
            window.openAuthModal();
            return;
        }

        if (!isMarketOpenNow()) {
            showToast('현재 장 마감 시간(00:00~05:59)입니다. 상단의 [야간 강제개장]을 켜거나 06:00 이후에 거래해 주세요.', 'error');
            return;
        }

        const acc = allAccounts[currentAccountKey];
        const st = stocks[selectedStockId];
        const limits = getPriceLimits(st.prevClose);
        const orderPrice = Number(document.getElementById('inputOrderPrice').value);
        const orderQty = Number(document.getElementById('inputOrderQty').value);

        if (orderQty <= 0 || orderPrice <= 0) {
            showToast('올바른 주문 단가와 수량을 입력해 주세요.', 'error');
            return;
        }

        if (orderPrice > limits.upper || orderPrice < limits.lower) {
            showToast(`상한가(${limits.upper.toLocaleString()}원) ~ 하한가(${limits.lower.toLocaleString()}원) 범위 내에서만 주문할 수 있습니다.`, 'error');
            return;
        }

        if (currentTradeSide === 'BUY') {
            if (orderPrice < st.price) {
                showToast(`즉시 체결을 위해 현재가(${st.price.toLocaleString()}원) 이상으로 단가를 입력해 주세요.`, 'error');
                return;
            }
            const cost = st.price * orderQty;
            if (acc.cash < cost) {
                showToast('주문 가능 현금이 부족합니다!', 'error');
                return;
            }
            acc.cash -= cost;
            const prev = acc.holdings[selectedStockId] || { qty: 0, avgPrice: 0 };
            const newQty = prev.qty + orderQty;
            const newAvg = ((prev.qty * prev.avgPrice) + cost) / newQty;
            acc.holdings[selectedStockId] = { qty: newQty, avgPrice: newAvg };

            showToast(`[매수 체결] ${st.name} ${orderQty.toLocaleString()}주 (${st.price.toLocaleString()}원)`, 'buy');
        } else {
            if (orderPrice > st.price) {
                showToast(`즉시 체결을 위해 현재가(${st.price.toLocaleString()}원) 이하로 단가를 입력해 주세요.`, 'error');
                return;
            }
            const prev = acc.holdings[selectedStockId];
            if (!prev || prev.qty < orderQty) {
                showToast('매도할 보유 주식이 부족합니다!', 'error');
                return;
            }
            acc.cash += st.price * orderQty;
            prev.qty -= orderQty;
            if (prev.qty === 0) prev.avgPrice = 0;

            showToast(`[매도 체결] ${st.name} ${orderQty.toLocaleString()}주 (${st.price.toLocaleString()}원)`, 'sell');
        }

        refreshAllViews();
        await saveAccountToCloud(currentAccountKey);
    };

    window.handleLoginOrRegister = async function() {
        const nickname = document.getElementById('authNickname').value.trim();
        const password = document.getElementById('authPassword').value.trim();

        if (!nickname) {
            showToast('닉네임을 입력해 주세요! (예: 나, 아빠)', 'error');
            return;
        }
        if (!password) {
            showToast('비밀번호를 입력해 주세요!', 'error');
            return;
        }

        const existing = allAccounts[nickname];
        if (existing) {
            if (existing.password !== password) {
                showToast('비밀번호가 일치하지 않습니다!', 'error');
                return;
            }
            currentAccountKey = nickname;
            showToast(`반갑습니다, [${nickname}]님! 저장된 계좌를 불러왔습니다.`, 'info');
        } else {
            // 신규 계정 생성
            allAccounts[nickname] = {
                nickname,
                password,
                cash: INITIAL_CASH,
                holdings: {}
            };
            currentAccountKey = nickname;
            await saveAccountToCloud(nickname);
            showToast(`[${nickname}] 새 계정이 생성되었습니다! (초기자금 5,000만 원 지급)`, 'buy');
        }

        window.closeAuthModal();
        refreshAllViews();
    };

    window.toggleNightMarket = function(checked) {
        nightMarketOverride = checked;
        updateClockAndMarketStatus();
        showToast(checked ? '야간 강제개장 모드가 켜졌습니다.' : '기본 거래시간(06:00~23:59) 모드로 전환되었습니다.', 'info');
    };

        modal.classList.add('flex');
    };

    window.closeBroadcastModal = function() {
        const modal = document.getElementById('broadcastModal');
        modal.classList.add('hidden');
        modal.classList.remove('flex');
    };

    window.triggerBroadcastUpdate = async function() {
        const note = document.getElementById('updateNoteInput').value.trim() || '새로운 버전 업데이트';
        const nextVer = (localSystemVersion || 1) + 1;
        localSystemVersion = nextVer;

        if (isFirebaseReady && auth.currentUser) {
            try {
                const configRef = doc(db, 'artifacts', appId, 'public', 'data', 'system', 'config');
                await setDoc(configRef, {
                    version: nextVer,
                    note: note,
                    updatedAt: Date.now()
                });
                showToast(`v${nextVer} 업데이트 알림이 모든 참가자 화면에 발송되었습니다!`, 'buy');
            } catch (e) {
                showToast('업데이트 알림 발송 중 오류가 발생했습니다.', 'error');
            }
        } else {
            // 로컬 테스트용 배너 즉시 표시
            document.getElementById('newVerText').textContent = `v${nextVer} - ${note}`;
            const banner = document.getElementById('updateBanner');
            banner.classList.remove('hidden');
            banner.classList.add('flex');
        }
        window.closeBroadcastModal();
    };

    window.applySystemUpdate = function() {
        window.location.reload();
    };

    function showToast(msg, type = 'info') {
        const wrap = document.getElementById('toastWrap');
        const el = document.createElement('div');
        let style = 'bg-slate-900 border-slate-700 text-white';
        if (type === 'buy') style = 'bg-[#241217] border-red-500/60 text-red-100';
        if (type === 'sell') style = 'bg-[#111a2e] border-blue-500/60 text-blue-100';
        if (type === 'error') style = 'bg-amber-950 border-amber-500/70 text-amber-100';

        el.className = `px-4 py-3 rounded-xl border shadow-2xl text-xs font-bold transition-all duration-300 ${style}`;
        el.textContent = msg;
        wrap.appendChild(el);
        setTimeout(() => {
            el.classList.add('opacity-0', 'translate-y-2');
            setTimeout(() => el.remove(), 300);
        }, 2800);
    }
